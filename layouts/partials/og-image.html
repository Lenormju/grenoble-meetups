{{- /* Retourne l'URL absolue de la card sociale (1200×630) de la page reçue. L'image est composée au build : assets/og/card-bg.png (dégradé, accent ambre et domaine, identiques partout) + le texte variable via images.Text. Hugo nomme le fichier d'après un hash de son contenu, donc modifier un événement produit une nouvelle URL, ce qui contourne les caches LinkedIn et Google. C'est l'approximation qui remplace les métriques de police absentes des templates : seul réglage à ajuster si un titre déborde. */ -}} {{- $advance := 0.55 -}} {{- $titleZone := 230 -}} {{- $subMax := {{- if .IsPage -}} {{- $frDays := slice "Dimanche" "Lundi" "Mardi" "Mercredi" "Jeudi" "Vendredi" "Samedi" -}} {{- $end := "" -}} {{- with .Params.endDate -}}{{- $end = time . -}}{{- end -}} {{- if $end -}} {{- /* Événement sur plusieurs jours : la seule date de début serait trompeuse. */ -}} {{- if eq ($end.Format "2006-01") (.Date.Format "2006-01") -}} {{- $meta = printf "%d → %d %s %d" .Date.Day $end.Day (partial "fr-month-name.html" .Date) .Date.Year -}} {{- else -}} {{- $meta = printf "%d %s → %d %s %d" .Date.Day (partial "fr-month-name.html" .Date) $end.Day (partial with .Params.time -}} {{- $meta = printf "%s · %s" $meta . -}} {{- end -}} {{- end -}} {{- $meta = upper $meta -}} {{- /* Le rendu de texte de Go ne gère pas les emoji couleur : on les retire du $title = trim (replaceRE `\s+` " " $title) " " -}} {{- $names := slice -}} {{- range $slug := .Params.groups -}} {{- with site.GetPage (printf "/groups/%s" $slug) -}} {{- $names = $names | append .Title -}} {{- end -}} {{- end -}} {{- $sub = delimit $names " · " -}} {{- /* Certains événements portent le nom de leur groupe : ne pas le répéter. */ -}} {{- if eq $sub $title -}}{{- $sub = "" -}}{{- end -}} {{- else if .IsSection -}} {{- if not .Parent.IsHome -}} {{- $count := len .RegularPages -}} {{- $meta = "MEETUPS TECH À GRENOBLE" -}} {{- $title = .Title -}} {{- $sub = cond (eq $count 0) "Aucun événement" (printf "%d événement%s" $count (cond (gt $count 1) "s" "")) -}} {{- end -}} {{- end -}} {{- end -}} {{- /* Taille de titre adaptative : on retient la plus grande qui tient dans
